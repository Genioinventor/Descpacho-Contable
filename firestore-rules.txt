rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // ─── Colección: usuarios ────────────────────────────────
    match /usuarios/{documento} {
      allow read:   if true;
      allow delete: if true;

      // Crear: requiere nombre y rfc
      allow create: if
        request.resource.data.keys().hasAll(['nombre', 'rfc']) &&
        request.resource.data.nombre is string &&
        request.resource.data.nombre.size() > 0 &&
        request.resource.data.nombre.size() <= 255 &&
        request.resource.data.rfc is string &&
        request.resource.data.rfc.size() <= 20;

      // Actualizar: mismo requisito
      allow update: if
        request.resource.data.keys().hasAll(['nombre', 'rfc']) &&
        request.resource.data.nombre is string &&
        request.resource.data.nombre.size() > 0 &&
        request.resource.data.nombre.size() <= 255 &&
        request.resource.data.rfc is string &&
        request.resource.data.rfc.size() <= 20;

      // Escritura generica (set con merge) — cubre merge parcial
      allow write: if true;
    }

    // ─── Colección: tickets ─────────────────────────────────
    match /tickets/{ticketId} {
      allow read:   if true;
      allow delete: if true;

      // Crear: requiere folio y total como minimo
      allow create: if
        request.resource.data.keys().hasAll(['folio', 'total']) &&
        request.resource.data.folio is number &&
        request.resource.data.total is number;

      // Actualizar: folio y total deben seguir presentes
      allow update: if
        request.resource.data.keys().hasAll(['folio', 'total']) &&
        request.resource.data.folio is number &&
        request.resource.data.total is number;

      // Escritura generica (set / set-merge) — permite merge parcial
      allow write: if true;
    }

  }
}
